# DealWell Affiliate Storefront — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Drive-managed affiliate storefront at dealwell.pro — a static "Warm & Editorial" site of product cards whose content is kept in a Google Sheet + Drive media folder and published automatically on a schedule.

**Architecture:** A static frontend (`site/`) renders from a generated `products.json` + local `media/`. A Node.js build tool (`sync/`) reads the Google Sheet (Sheets API) and the Drive media folder (Drive API) using a single Google API key, matches media to products by the leading `#`, and writes `site/products.json` + `site/media/`. A GitHub Actions workflow runs the build hourly and FTP-deploys `site/` to Hostinger. No server-side code; the scheduler is GitHub Actions (free, Claude-independent).

**Tech Stack:** Node.js 24 (built-in `node --test`, global `fetch`, ES modules — no runtime npm deps), vanilla HTML/CSS/JS frontend, GitHub Actions (cron), Hostinger shared hosting (FTP deploy), Google Sheets API v4 + Google Drive API v3 (API key).

## Global Constraints

- **Language / copy:** English (US) for all user-facing text.
- **Button copy:** exactly `View on <Channel> →` (e.g. `View on Amazon →`).
- **Palette (Option C — Warm & Editorial):** background `#fbf7f1`, text `#1f1b17`, gold accent `#c08a4a`, card surface `#ffffff`, muted `#6b5d4d`. Serif (`Georgia, 'Times New Roman', serif`) for logo + product names; system sans-serif for body.
- **Outbound links:** open in new tab with `target="_blank" rel="nofollow sponsored noopener"`.
- **Affiliate disclosure:** footer must contain `As an affiliate, DealWell earns from qualifying purchases.`
- **Spreadsheet columns (order):** `#`, `Name`, `Category`, `Description`, `Channel`, `Link`, `Show`. Sheet/tab name for data: `Products`.
- **Media naming:** filename starts with the product `#` then a separator (`.`/`-`/`_`), e.g. `1.1.jpg`. First (natural-sorted) file = cover; all = gallery. Images: `.jpg .jpeg .png .webp .gif`; videos: `.mp4 .mov .webm`.
- **Row validity:** a row missing `Name`, `Channel`, or `Link` is skipped. `Show` ∈ {`no`,`false`,`0`,`hide`} (case-insensitive) → hidden; otherwise shown. Duplicate `#` → first wins.
- **Sync cadence:** GitHub Actions cron hourly (`0 * * * *`) + manual `workflow_dispatch`.
- **No runtime npm dependencies** in `sync/` (use built-in `fetch`, `node:fs`, `node:path`). Dev-only deps are allowed if ever needed.
- **Node version:** 24.x (matches local + Actions `node-version: 24`).

```js
// site/assets/catalog-helpers.js
export function buttonLabel(channel) {
  return `View on ${channel} →`;
}

export function coverOf(product) {
  return product.media && product.media.length ? product.media[0] : null;
}

export function withAllPicks(categories) {
  const seen = new Set();
  const out = ['All Picks'];
  for (const c of categories) { if (c && !seen.has(c)) { seen.add(c); out.push(c); } }
  return out;
}

export function filterProducts(products, { category = 'All Picks', query = '' } = {}) {
  const q = query.trim().toLowerCase();
  return products.filter(p => {
    const catOk = category === 'All Picks' || p.category === category;
    const qOk = q === '' || (p.name || '').toLowerCase().includes(q);
    return catOk && qOk;
  });
}

export function nextIndex(i, len) { return len ? (i + 1) % len : 0; }
export function prevIndex(i, len) { return len ? (i - 1 + len) % len : 0; }
```

```js
const HIDDEN = new Set(['no','false','0','hide']);
const cell = (row, i) => (row[i] == null ? '' : String(row[i]).trim());

export function parseProducts(rows) {
  const products = [];
  const categories = [];
  const seenCat = new Set();
  const seenId = new Set();
  const skipped = [];

  for (let r = 1; r < rows.length; r++) {
    const row = rows[r] || [];
    const idRaw = cell(row, 0);
    const name = cell(row, 1);
    const category = cell(row, 2);
    const description = cell(row, 3);
    const channel = cell(row, 4);
    const link = cell(row, 5);
    const show = cell(row, 6).toLowerCase();

    if (!idRaw && !name && !channel && !link) continue; // blank row

    if (HIDDEN.has(show)) continue;                      // hidden, not an error
    if (!name)    { skipped.push({ row: r+1, reason: 'missing Name' }); continue; }
    if (!channel) { skipped.push({ row: r+1, reason: 'missing Channel' }); continue; }
    if (!link)    { skipped.push({ row: r+1, reason: 'missing Link' }); continue; }

    const id = Number.parseInt(idRaw, 10);
    if (!Number.isFinite(id)) { skipped.push({ row: r+1, reason: `invalid # "${idRaw}"` }); continue; }
    if (seenId.has(id)) { skipped.push({ row: r+1, reason: `duplicate # ${id}` }); continue; }
    seenId.add(id);

    if (category && !seenCat.has(category)) { seenCat.add(category); categories.push(category); }
    products.push({ id, name, category, description, channel, link });
  }
  return { products, categories, skipped };
}
```

```js
// sync/src/matchMedia.js
const IMAGE = new Set(['jpg','jpeg','png','webp','gif']);
const VIDEO = new Set(['mp4','mov','webm']);

const ext = (name) => { const i = name.lastIndexOf('.'); return i < 0 ? '' : name.slice(i+1).toLowerCase(); };

export function mediaType(name) {
  const e = ext(name);
  if (IMAGE.has(e)) return 'image';
  if (VIDEO.has(e)) return 'video';
  return null;
}

export function productIdFromFilename(name) {
  const m = /^(\d+)[._-]/.exec(name);
  return m ? Number.parseInt(m[1], 10) : null;
}

// natural compare so 1.2 < 1.10
function naturalCompare(a, b) {
  return a.localeCompare(b, undefined, { numeric: true, sensitivity: 'base' });
}

export function matchMedia(products, files, { mediaPath = 'media' } = {}) {
  const byId = new Map();
  for (const f of files) {
    const type = mediaType(f.name);
    const pid = productIdFromFilename(f.name);
    if (type == null || pid == null) continue;
    if (!byId.has(pid)) byId.set(pid, []);
    byId.get(pid).push({ type, fileId: f.id, fileName: f.name });
  }
  for (const list of byId.values()) list.sort((x, y) => naturalCompare(x.fileName, y.fileName));

  return products.map(p => ({
    ...p,
    media: (byId.get(p.id) || []).map(m => ({
      type: m.type, src: `${mediaPath}/${m.fileName}`, fileId: m.fileId, fileName: m.fileName,
    })),
  }));
}
```

- [ ] **Step 1: Write `sync/src/google.js`**

```js
// sync/src/google.js
import { mkdir, writeFile } from 'node:fs/promises';
import { dirname } from 'node:path';

const SHEETS = 'https://sheets.googleapis.com/v4/spreadsheets';
const DRIVE = 'https://www.googleapis.com/drive/v3';

async function getJson(url) {
  const res = await fetch(url);
  if (!res.ok) throw new Error(`GET ${url.split('?')[0]} -> ${res.status} ${await res.text()}`);
  return res.json();
}

export async function fetchSheetRows({ sheetId, tab = 'Products', apiKey }) {
  const url = `${SHEETS}/${sheetId}/values/${encodeURIComponent(tab)}?key=${apiKey}`;
  const data = await getJson(url);
  return data.values || [];
}

export async function listDriveFolder({ folderId, apiKey }) {
  const files = [];
  let pageToken = '';
  do {
    const q = encodeURIComponent(`'${folderId}' in parents and trashed = false`);
    const fields = encodeURIComponent('nextPageToken, files(id,name,mimeType,modifiedTime)');
    const url = `${DRIVE}/files?q=${q}&key=${apiKey}&fields=${fields}&pageSize=1000`
      + (pageToken ? `&pageToken=${pageToken}` : '');
    const data = await getJson(url);
    files.push(...(data.files || []));
    pageToken = data.nextPageToken || '';
  } while (pageToken);
  return files;
}

export async function downloadDriveFile({ fileId, apiKey, destPath }) {
  const url = `${DRIVE}/files/${fileId}?alt=media&key=${apiKey}`;
  const res = await fetch(url);
  if (!res.ok) throw new Error(`download ${fileId} -> ${res.status}`);
  await mkdir(dirname(destPath), { recursive: true });
  const buf = Buffer.from(await res.arrayBuffer());
  await writeFile(destPath, buf);
}
```

## Task 8: `build.js` orchestrator — generate products.json + media

**Files:**
- Create: `sync/build.js`

**Interfaces:**
- Consumes: `parseProducts`, `matchMedia`, `buildCatalog`, `google.js`.
- Reads env: `GOOGLE_API_KEY`, `SHEET_ID`, `DRIVE_FOLDER_ID`; optional `SITE_DIR` (default `site`), `SHEET_TAB` (default `Products`).
- Produces: writes `${SITE_DIR}/products.json` and downloads referenced media into `${SITE_DIR}/media/` (only files not already present with the same name; removes local media no longer referenced). Logs a summary. Exits non-zero on fatal error.

- [ ] **Step 1: Write `sync/build.js`**

```js
// sync/build.js
import { readdir, mkdir, writeFile, rm, access } from 'node:fs/promises';
import { join } from 'node:path';
import { parseProducts } from './src/parseProducts.js';
import { matchMedia } from './src/matchMedia.js';
import { buildCatalog } from './src/buildCatalog.js';
import { fetchSheetRows, listDriveFolder, downloadDriveFile } from './src/google.js';

function requireEnv(name) {
  const v = process.env[name];
  if (!v) { console.error(`Missing env ${name}`); process.exit(1); }
  return v;
}
const exists = (p) => access(p).then(() => true).catch(() => false);

async function main() {
  const apiKey = requireEnv('GOOGLE_API_KEY');
  const sheetId = requireEnv('SHEET_ID');
  const folderId = requireEnv('DRIVE_FOLDER_ID');
  const siteDir = process.env.SITE_DIR || 'site';
  const tab = process.env.SHEET_TAB || 'Products';
  const mediaDir = join(siteDir, 'media');

  const rows = await fetchSheetRows({ sheetId, tab, apiKey });
  const parsed = parseProducts(rows);
  const files = await listDriveFolder({ folderId, apiKey });
  const withMedia = matchMedia(parsed.products, files);

  // Download referenced media that is missing locally.
  await mkdir(mediaDir, { recursive: true });
  const referenced = new Set();
  let downloaded = 0;
  for (const p of withMedia) {
    for (const m of p.media) {
      referenced.add(m.fileName);
      const dest = join(mediaDir, m.fileName);
      if (!(await exists(dest))) { await downloadDriveFile({ fileId: m.fileId, apiKey, destPath: dest }); downloaded++; }
    }
  }
  // Remove local media no longer referenced (keep .gitkeep).
  let removed = 0;
  for (const name of await readdir(mediaDir)) {
    if (name === '.gitkeep') continue;
    if (!referenced.has(name)) { await rm(join(mediaDir, name)); removed++; }
  }

  const catalog = buildCatalog({ products: withMedia, categories: parsed.categories }, new Date().toISOString());
  await writeFile(join(siteDir, 'products.json'), JSON.stringify(catalog, null, 2));

  console.log(`published=${catalog.products.length} skipped=${parsed.skipped.length} media_downloaded=${downloaded} media_removed=${removed}`);
  for (const s of parsed.skipped) console.log(`  skip row ${s.row}: ${s.reason}`);
}

main().catch((e) => { console.error(e); process.exit(1); });
```
